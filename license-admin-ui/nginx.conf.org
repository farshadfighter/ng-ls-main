# nginx config for the built admin UI (see Dockerfile.prod).
# Serves the SPA over HTTPS and proxies API calls to the license server container.

# Plain HTTP exists only to bounce clients to HTTPS. Nothing — neither the admin
# UI nor /api — is served over cleartext, so a client that forgets the scheme
# cannot leak an admin password or a license key on the wire before finding out.
#
# 308 rather than 301: a 301 permits clients to replay the request as GET, which
# would quietly turn a POST /api/licenses/activate into a bodyless GET. A 308
# preserves method and body, so the retry over TLS is the same request.
#
# $host (not a hard-coded name) keeps whatever address the client used — this
# server is reached by IP today and by license.ngcorion.local elsewhere.
server {
    listen 80;
    return 308 https://$host$request_uri;
}

server {
    # HTTPS for the License Server API and the admin UI. This host is internal
    # and is reached by IP (no public subdomain, so no ACME/public CA is
    # possible), which is why the certificate is self-signed with an IP SAN —
    # generate it with ./scripts/gen-self-signed-cert.sh before starting.
    listen 443 ssl;

    ssl_certificate     /etc/nginx/certs/license-server.crt;
    ssl_certificate_key /etc/nginx/certs/license-server.key;

    # TLS 1.2 is the floor: the app server's license client talks through
    # python-requests/OpenSSL, which negotiates 1.3 here, and nothing older than
    # 1.2 has any client on this network.
    ssl_protocols       TLSv1.2 TLSv1.3;
    ssl_ciphers         HIGH:!aNULL:!MD5;
    ssl_session_cache   shared:SSL:10m;
    ssl_session_timeout 1h;

    root /usr/share/nginx/html;
    index index.html;

    # SPA fallback: deep links resolve in the browser, not on disk.
    location / {
        try_files $uri $uri/ /index.html;
    }

    # Admin + license API. `license-server` resolves on the compose network.
    location /api/ {
        proxy_pass http://license-server:8001/api/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        # uvicorn runs with --proxy-headers and reads X-Forwarded-For. Without
        # this header every proxied request looks like it came from the nginx
        # container, so the per-IP rate limiter throttles all admin users
        # against one shared bucket.
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        # Now always "https": the app is behind TLS termination here, and
        # anything the API builds from the scheme must say so.
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # Mirrors the Vite dev proxy, which forwards /health too (src/api/client.js
    # exposes serverHealth()). Without it this path would hit the SPA fallback.
    location /health {
        proxy_pass http://license-server:8001/health;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
