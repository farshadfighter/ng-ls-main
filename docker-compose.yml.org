# Standalone License Server stack: Postgres + Redis + API + admin UI.
#
# Configuration comes from a root .env (see .env.example) and is passed to the
# containers as real environment variables. Nothing is read from a .env baked
# into an image — that file is .dockerignore'd precisely so secrets stay out of
# the image layers.
#
#   cp .env.example .env  &&  edit  &&  docker compose up -d --build

services:
  postgres:
    image: postgres:18-alpine
    container_name: ng-postgres
    restart: unless-stopped
    environment:
      POSTGRES_USER: ${POSTGRES_USER:-user}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-password}
      POSTGRES_DB: ${POSTGRES_DB:-mydb}
    volumes:
      # postgres:18 sets PGDATA to /var/lib/postgresql/18/docker and declares
      # VOLUME /var/lib/postgresql, so the mount belongs on the parent dir.
      # (On postgres:17 and older this must be /var/lib/postgresql/data.)
      - pgdata:/var/lib/postgresql
    networks:
      - app-network
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER:-user} -d ${POSTGRES_DB:-mydb}"]
      interval: 5s
      timeout: 5s
      retries: 5

  redis:
    image: redis:8-alpine
    container_name: ng-redis
    restart: unless-stopped
    command: redis-server --save 60 1 --loglevel warning
    volumes:
      - redisdata:/data
    networks:
      - app-network
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 5s
      timeout: 3s
      retries: 5

  backend:
    build:
      context: license_server
      dockerfile: Dockerfile
    container_name: license-server
    restart: unless-stopped
    # NOT published to the host any more. The License Server API is served over
    # HTTPS on :443 by the `front` nginx, which reaches this container on the
    # internal compose network. Publishing 8001 as well would keep an
    # unencrypted copy of the same API listening on the host.
    expose:
      - "8001"
    environment:
      - DATABASE_URL=postgresql://${POSTGRES_USER:-user}:${POSTGRES_PASSWORD:-password}@postgres:5432/${POSTGRES_DB:-mydb}
      # app/core/config.py reads REDIS_HOST/REDIS_PORT/REDIS_DB — there is no
      # REDIS_URL setting. Passing a URL here (as this file used to) left the
      # host at its "localhost" default, so the API could never reach Redis and
      # rate limiting silently degraded to "allow every request".
      - REDIS_HOST=redis
      - REDIS_PORT=6379
      - REDIS_DB=0
      # Required: Settings() has no default for SECRET_KEY, so the API refuses
      # to import without it. Fail here, with a readable message, rather than in
      # a container restart loop.
      - SECRET_KEY=${SECRET_KEY:?set SECRET_KEY in .env (openssl rand -hex 32)}
      - ADMIN_USERNAME=${ADMIN_USERNAME:-admin}
      - ADMIN_PASSWORD=${ADMIN_PASSWORD:?set ADMIN_PASSWORD in .env}
      - RATE_LIMIT_PER_MINUTE=${RATE_LIMIT_PER_MINUTE:-60}
      - JWT_EXPIRE_HOURS=${JWT_EXPIRE_HOURS:-24}
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    networks:
      - app-network
    healthcheck:
      # The image has no curl/wget; python is guaranteed to be there.
      test: ["CMD", "python", "-c", "import urllib.request,sys; sys.exit(0 if urllib.request.urlopen('http://127.0.0.1:8001/health', timeout=3).status == 200 else 1)"]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 20s

  front:
    build:
      context: license-admin-ui
      dockerfile: Dockerfile.prod
    container_name: ng-frontend
    restart: unless-stopped
    ports:
      # Plain HTTP: serves nothing but a 308 redirect to HTTPS (nginx.conf).
      - "${FRONTEND_PORT:-80}:80"
      # HTTPS: this is the License Server API's published address now
      # (https://<this-host>), and the only way the admin UI and /api are served.
      - "${HTTPS_PORT:-443}:443"
    volumes:
      # Self-signed cert + key, generated by ./scripts/gen-self-signed-cert.sh.
      # Mounted read-only rather than baked into the image so the private key
      # stays out of every image layer. nginx will refuse to start if these are
      # missing — run the script before `docker compose up`.
      - ./certs:/etc/nginx/certs:ro
    depends_on:
      backend:
        # nginx resolves `license-server` when it starts and fails hard with
        # "host not found in upstream" if the backend is not up yet, so wait for
        # healthy rather than merely started.
        condition: service_healthy
    networks:
      - app-network
    healthcheck:
      # Probes the HTTPS listener, not port 80 — port 80 only ever returns a
      # redirect now, so a 2xx there would prove nothing about the service.
      # --cacert (not -k) means an expired or mismatched certificate makes this
      # container unhealthy instead of quietly serving a broken TLS listener.
      # The cert carries 127.0.0.1 in its SAN for exactly this probe.
      test: ["CMD", "curl", "-fsS", "--cacert", "/etc/nginx/certs/license-server.crt",
             "-o", "/dev/null", "https://127.0.0.1/"]
      interval: 10s
      timeout: 5s
      retries: 5
      start_period: 10s

volumes:
  pgdata:
    driver: local
  redisdata:
    driver: local

networks:
  app-network:
    driver: bridge
